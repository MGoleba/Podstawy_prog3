## 31. ŚCIĄGA — PRZYDATNE ELEMENTY Z CAŁEGO MATERIAŁU

---------------------------------------------------------------------------
print()            — wyświetla dane w terminalu.
input()            — czyta tekst wpisany przez użytkownika.
int(...)           — konwertuje poprawną wartość na liczbę całkowitą.
float(...)         — konwertuje poprawną wartość na liczbę dziesiętną.
str(...)           — konwertuje wartość na tekst.
type(...)          — zwraca typ podanej wartości.
len(...)           — podaje liczbę elementów, np. znaków napisu.
round(liczba, 2)   — zaokrągla liczbę do 2 miejsc po przecinku.
import math        — udostępnia funkcje i stałe matematyczne.
math.sqrt(x)       — pierwiastek kwadratowy z x.
math.pi            — liczba pi.
import random      — udostępnia funkcje do losowania.
random.randint(a,b)— losuje liczbę całkowitą od a do b (włącznie).
range(5)           — tworzy zakres 0, 1, 2, 3, 4.
for ... in ...:    — wykonuje blok kodu wielokrotnie.
upper()            — zwraca tekst z wielkimi literami.
lower()            — zwraca tekst z małymi literami.
count('e')         — zlicza wystąpienia małej litery e.
find('Lord')       — zwraca pozycję początku szukanego tekstu lub -1.
ord('A')           — kod Unicode pojedynczego znaku.
chr(65)            — znak Unicode o podanym kodzie.
and                — logiczne ORAZ (oba warunki muszą być prawdziwe).
or                 — logiczne LUB (wystarczy jeden prawdziwy warunek).
not                — logiczne NIE (odwraca True i False).
=                  — przypisanie wartości do zmiennej.
==                 — porównanie równości wartości.
!=                 — porównanie różności wartości.
**                 — potęgowanie.
/                  — dzielenie zwykłe.
//                 — dzielenie z zaokrągleniem w dół.
%                  — reszta z dzielenia.
# komentarz        — informacja dla człowieka, pomijana przez Pythona.

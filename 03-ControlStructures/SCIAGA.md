ZAPIS                        ZNACZENIE
if warunek:                  wykonaj blok, jeśli warunek daje True.
elif inny_warunek:           sprawdź kolejny warunek, gdy poprzednie są False.
else:                        wykonaj blok, gdy wcześniejsze warunki są False.
>=                           większe lub równe.
<=                           mniejsze lub równe.
==                           równe (porównanie, NIE przypisanie).
!=                           różne od.
and                          oba warunki muszą być prawdziwe.
or                           przynajmniej jeden warunek musi być prawdziwy.
not                          odwraca wartość logiczną.
%                            reszta z dzielenia (np. sprawdzanie parzystości).
for element in tekst:        przejdź przez kolejne znaki napisu.
for i in range(5):           wykonaj blok dla i = 0, 1, 2, 3, 4.
for i in range(1, 6):        wykonaj blok dla i = 1, 2, 3, 4, 5.
while warunek:               powtarzaj blok, dopóki warunek daje True.
while True:                  pętla bez naturalnego zakończenia.
continue                     pomiń resztę bieżącej iteracji pętli.
break                        natychmiast opuść pętlę.
total = total + i            dodaj i do dotychczasowej sumy.
total += i                   krótszy zapis dokładnie tego samego.
print(...)                   wyświetl wartości w terminalu.
input(...)                   pobierz tekst wpisany przez użytkownika.
int(...)                     konwertuj na liczbę całkowitą.
f'Tekst {zmienna}'           wstaw wartość zmiennej do tekstu.
4 spacje przed instrukcją    zaznacz, że instrukcja należy do danego bloku.
8 spacji                     zagnieżdżony blok wewnątrz innego bloku.
range(start, stop)           zawiera start, NIE zawiera stop.
True / False                 logiczna prawda / fałsz.
